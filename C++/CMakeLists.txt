cmake_minimum_required(VERSION 3.18)
project(mhec VERSION 0.1 LANGUAGES CXX)
include(CTest)
option(MHEC_BUILD_EXAMPLES "Build the hardware-free example" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Prefer an installed OpenCV. The existing Windows SDK subset has import
# libraries but no runtime DLLs; use it without its incomplete package config.
if(WIN32 AND MSVC AND NOT OpenCV_DIR)
  find_library(MHEC_OPENCV_RELEASE opencv_world4100 PATHS "${CMAKE_CURRENT_SOURCE_DIR}/dependence/opencv/x64/vc16/lib" NO_DEFAULT_PATH REQUIRED)
  find_library(MHEC_OPENCV_DEBUG opencv_world4100d PATHS "${CMAKE_CURRENT_SOURCE_DIR}/dependence/opencv/x64/vc16/lib" NO_DEFAULT_PATH REQUIRED)
  add_library(mhec_opencv INTERFACE)
  target_include_directories(mhec_opencv INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/dependence/opencv/include")
  target_link_libraries(mhec_opencv INTERFACE optimized "${MHEC_OPENCV_RELEASE}" debug "${MHEC_OPENCV_DEBUG}")
else()
  find_package(OpenCV REQUIRED COMPONENTS core)
  add_library(mhec_opencv INTERFACE)
  target_include_directories(mhec_opencv INTERFACE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(mhec_opencv INTERFACE ${OpenCV_LIBS})
endif()
add_library(mhec src/calibration.cpp src/estimators.cpp src/math.cpp)
target_include_directories(mhec PUBLIC include dependence/eigen-3.4.0)
target_link_libraries(mhec PUBLIC mhec_opencv)
if(MSVC)
  target_compile_options(mhec PRIVATE /utf-8)
endif()
if(MHEC_BUILD_EXAMPLES)
  add_executable(demo_calibration examples/simulation/main.cpp)
  target_link_libraries(demo_calibration PRIVATE mhec)
endif()
if(BUILD_TESTING)
  add_executable(test_core tests/test_core.cpp)
  target_link_libraries(test_core PRIVATE mhec)
  add_test(NAME core COMMAND test_core "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/before_extraction.txt")
endif()
